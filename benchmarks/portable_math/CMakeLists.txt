cmake_minimum_required(VERSION 3.17)
project(PortableMathBenchmark LANGUAGES C CXX)
get_filename_component(PINEFORGE_HPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE
    CACHE)
set(PINEFORGE_HPO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PINEFORGE_HPO_BUILD_ENGINE_ADAPTER OFF CACHE BOOL "" FORCE)
set(PINEFORGE_HPO_BUILD_NATIVE_CLI OFF CACHE BOOL "" FORCE)
add_subdirectory("${PINEFORGE_HPO_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/hpo")
add_executable(proposal_benchmark proposal_benchmark.cpp)
target_link_libraries(proposal_benchmark PRIVATE PineForgeHPO::core)
target_compile_features(proposal_benchmark PRIVATE cxx_std_17)
target_compile_options(proposal_benchmark PRIVATE -ffp-contract=off)
add_executable(compatibility_probe compatibility_probe.cpp)
target_link_libraries(compatibility_probe PRIVATE PineForgeHPO::core)
target_compile_features(compatibility_probe PRIVATE cxx_std_17)
target_compile_options(compatibility_probe PRIVATE -ffp-contract=off)
